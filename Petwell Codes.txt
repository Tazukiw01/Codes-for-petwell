            string password = txtPassword.Text.Trim();
            if (string.IsNullOrWhiteSpace(login) || string.IsNullOrWhiteSpace(password))
            {
                MessageBox.Show("Please enter Email/Phone Number and Password.");
                return;
            }
            try
            {
                using (SqlConnection con = new SqlConnection(connStr))
                {
                    con.Open();

                    string sql = @"
    SELECT UserId, Role, IsSuperAdmin
    FROM table name 
    WHERE (PhoneNumber = @Login OR Email = @Login)
      AND Password = @Password;
";


                    using (SqlCommand cmd = new SqlCommand(sql, con))
                    {
                        cmd.Parameters.AddWithValue("@Login", login);
                        cmd.Parameters.AddWithValue("@Password", password);
                       
                        using (SqlDataReader reader = cmd.ExecuteReader())
                        {
                            if (!reader.Read())
                            {
                                MessageBox.Show("Wrong Email/Phone Number or Password.");
                                return;
                            }
                            int userId = Convert.ToInt32(reader["UserId"]);
                            string role = reader["Role"].ToString();
                            bool isSuperAdmin = Convert.ToBoolean(reader["IsSuperAdmin"]);
                            Session.CurrentUserId = userId;
                            Session.CurrentRole = role;
                            Session.IsSuperAdmin = isSuperAdmin;


                            reader.Close(); // close reader before next command
                            string loginType = login.Contains("@") ? "Email" : "Phone";
                            LogActivity(con, userId, "LOGIN", $"{role} logged in successfully using {loginType}");
                            this.Hide(); // hide login first

                            Form nextForm;
                            if (role == "Admin")
                            {
                                nextForm = new AdminForm();
                                nextForm.Show();
                                return;
                            }
                            else if (role == "Reception")
                            {
                                nextForm = new ReceptionForm();
                                nextForm.Show();
                                return;
                            }
                            else if (role == "User")
                            {
                                nextForm = new HomePage();
                                nextForm.Show();
                                return;
                            }
                            else
                            {
                                MessageBox.Show("Unknown role in database.");
                                this.Show();
                                return;
                            }

                        }
                    }
                }
            }
            catch (Exception ex)
            {
                MessageBox.Show("Login error: " + ex.Message);
            }
        }

   public partial class ForgotPasswordForm : BaseForm
   {
       private readonly ForgotPasswordRepository _repo = new ForgotPasswordRepository();

       public ForgotPasswordForm()
       {
           InitializeComponent();
           this.Load += ForgotPasswordForm_Load;
       }

       private void ForgotPasswordForm_Load(object sender, EventArgs e)
       {
           txtNewPassword.Visible = false;
           txtConfirmPassword.Visible = false;
           btnReset.Visible = false;

           txtNewPassword.UseSystemPasswordChar = true;
           txtConfirmPassword.UseSystemPasswordChar = true;
       }

       private void btnVerify_Click(object sender, EventArgs e)
       {
           string phoneNumber = txtPhone.Text.Trim();

           if (string.IsNullOrWhiteSpace(phoneNumber))
           {
               MessageBox.Show("Please enter your phone number.");
               return;
           }

           bool exists = _repo.PhoneExists(phoneNumber);

           if (!exists)
           {
               MessageBox.Show("Phone number not found.");
               return;
           }

           MessageBox.Show("Phone number verified. You can now reset your password.");

           txtPhone.ReadOnly = true;

           txtNewPassword.Visible = true;
           txtConfirmPassword.Visible = true;
           btnReset.Visible = true;

           txtNewPassword.Focus();
       }

       private void btnReset_Click(object sender, EventArgs e)
       {
           string phoneNumber = txtPhone.Text.Trim();
           string newPassword = txtNewPassword.Text.Trim();
           string confirmPassword = txtConfirmPassword.Text.Trim();

           if (string.IsNullOrWhiteSpace(newPassword) || string.IsNullOrWhiteSpace(confirmPassword))
           {
               MessageBox.Show("Please enter both password fields.");
               return;
           }

           if (newPassword != confirmPassword)
           {
               MessageBox.Show("Passwords do not match.");
               return;
           }

           bool success = _repo.ResetPassword(phoneNumber, newPassword);

           if (success)
           {
               MessageBox.Show("Password reset successfully.");

               LoginForm login = new LoginForm();
               login.Show();
               this.Hide();
           }
           else
           {
               MessageBox.Show("Failed to reset password.");
           }
       }

       private void btnBack_Click(object sender, EventArgs e)
       {
           LoginForm login = new LoginForm();
           login.Show();
           this.Hide();
       }

        private void LoadUsersAndLogs()
        {
            try
            {
                using (SqlConnection con = new SqlConnection(connStr))
                {
                    con.Open();
                    string usersQuery = @"
                SELECT 
                    ROW_NUMBER() OVER (ORDER BY UserId DESC) AS RowNum,
                    UserId,
                    Name,
                    Email,
                    PhoneNumber,
                    Password,
                    Role,
                    IsSuperAdmin
                FROM dbo.tableName;
            ";

                    using (SqlDataAdapter daUsers = new SqlDataAdapter(usersQuery, con))
                    {
                        DataTable dtUsers = new DataTable();
                        daUsers.Fill(dtUsers);

                        dgvUsers.DataSource = dtUsers;

                        dgvUsers.SelectionMode = DataGridViewSelectionMode.FullRowSelect;
                        dgvUsers.MultiSelect = false;
                        dgvUsers.AllowUserToAddRows = false;
                        dgvUsers.ReadOnly = true;

                        if (dgvUsers.Columns["Password"] != null)
                            dgvUsers.Columns["Password"].Visible = false;

                        if (dgvUsers.Columns["IsSuperAdmin"] != null)
                            dgvUsers.Columns["IsSuperAdmin"].Visible = false;

                        if (dgvUsers.Columns["UserId"] != null)
                            dgvUsers.Columns["UserId"].Visible = false;

                        if (dgvUsers.Columns["RowNum"] != null)
                        {
                            dgvUsers.Columns["RowNum"].HeaderText = "#";
                            dgvUsers.Columns["RowNum"].DisplayIndex = 0;
                            dgvUsers.Columns["RowNum"].Width = 50;
                        }

                        dgvUsers.ClearSelection();
                    }

                    string logsSql = @"
                SELECT TOP 200
                    a.ActivityId,
                    u.Name,
                    u.Role,
                    u.PhoneNumber,
                    a.ActivityType,
                    a.ActivityDetails,
                    a.CreatedAt
                FROM dbo.tableName a
                JOIN dbo.Tablename u ON u.UserId = a.UserId
                ORDER BY a.CreatedAt DESC;
            ";

                    using (SqlDataAdapter daLogs = new SqlDataAdapter(logsSql, con))
                    {
                        DataTable dtLogs = new DataTable();
                        daLogs.Fill(dtLogs);

                        dgvLogs.DataSource = dtLogs;

                        dgvLogs.SelectionMode = DataGridViewSelectionMode.FullRowSelect;
                        dgvLogs.MultiSelect = false;
                        dgvLogs.AllowUserToAddRows = false;
                        dgvLogs.ReadOnly = true;

                        dgvLogs.ClearSelection();
                    }
                }
            }
            catch (Exception ex)
            {
                MessageBox.Show("Error loading data: " + ex.Message);
            }

  private void btnDeleteUser_Click(object sender, EventArgs e)
  {
      if (dgvUsers.CurrentRow == null)
      {
          MessageBox.Show("Please select a user to delete.");
          return;
      }

      int userId = Convert.ToInt32(dgvUsers.CurrentRow.Cells["UserId"].Value);
      string role = dgvUsers.CurrentRow.Cells["Role"].Value.ToString();
      string name = dgvUsers.CurrentRow.Cells["Name"].Value.ToString();
      string phone = dgvUsers.CurrentRow.Cells["PhoneNumber"].Value.ToString();

      bool selectedIsSuperAdmin = Convert.ToBoolean(dgvUsers.CurrentRow.Cells["IsSuperAdmin"].Value);

      int adminId = Session.CurrentUserId;

      if (userId == adminId)
      {
          MessageBox.Show("You cannot delete your own account.", "Denied",
              MessageBoxButtons.OK, MessageBoxIcon.Warning);
          return;
      }

      if (selectedIsSuperAdmin)
      {
          MessageBox.Show("You cannot delete the Main Admin account.", "Denied",
              MessageBoxButtons.OK, MessageBoxIcon.Warning);
          return;
      }

      if (!Session.IsSuperAdmin && role == "Admin")
      {
          MessageBox.Show("Only the Main Admin can delete Admin accounts.", "Denied",
              MessageBoxButtons.OK, MessageBoxIcon.Warning);
          return;
      }

      DialogResult result = MessageBox.Show(
          $"Are you sure you want to delete this account?\n\nName: {name}\nPhone: {phone}\nRole: {role}",
          "Confirm Delete",
          MessageBoxButtons.YesNo,
          MessageBoxIcon.Warning);

      if (result != DialogResult.Yes)
          return;

      try
      {
          using (SqlConnection con = new SqlConnection(connStr))
          {
              con.Open();

              string deleteLogsSql = @"DELETE FROM dbo.tablename WHERE UserId = @UserId;";
              using (SqlCommand cmdLogs = new SqlCommand(deleteLogsSql, con))
              {
                  cmdLogs.Parameters.AddWithValue("@UserId", userId);
                  cmdLogs.ExecuteNonQuery();
              }

              string deleteUserSql = @"DELETE FROM dbo.tablename WHERE UserId = @UserId;";
              using (SqlCommand cmdUser = new SqlCommand(deleteUserSql, con))
              {
                  cmdUser.Parameters.AddWithValue("@UserId", userId);
                  int rows = cmdUser.ExecuteNonQuery();

                  if (rows == 0)
                  {
                      MessageBox.Show("User not found or already deleted.");
                      return;
                  }
              }
              string logSql = @"
          INSERT INTO dbo.tablename (UserId, ActivityType, ActivityDetails)
          VALUES (@AdminId, 'DELETE_USER', @Details);
      ";

              using (SqlCommand cmdLog = new SqlCommand(logSql, con))
              {
                  cmdLog.Parameters.AddWithValue("@AdminId", adminId);
                  cmdLog.Parameters.AddWithValue("@Details",
                      $"Admin deleted account: {name} (UserId: {userId}, Phone: {phone}, Role: {role})");
                  cmdLog.ExecuteNonQuery();
              }
          }

          MessageBox.Show("Account deleted successfully.");

          LoadUsersAndLogs();
      }
      catch (Exception ex)
      {
          MessageBox.Show("Delete error: " + ex.Message);
      }
  }

  private void LoadUsers(string searchText = "")
  {
      using (SqlConnection con = new SqlConnection(connStr))
      {
          con.Open();

          string usersSql = @"
      SELECT 
          ROW_NUMBER() OVER (ORDER BY UserId DESC) AS RowNum,
          UserId,
          Name,
          Email,
          PhoneNumber,
          Password,
          Role,
          IsSuperAdmin
      FROM dbo.tableName
      WHERE 
          (@Search = '' OR 
           Name LIKE '%' + @Search + '%' OR
           Email LIKE '%' + @Search + '%' OR
           PhoneNumber LIKE '%' + @Search + '%' OR
           Role LIKE '%' + @Search + '%' OR
           CAST(UserId AS NVARCHAR) LIKE '%' + @Search + '%')
      ORDER BY UserId DESC;
  ";

          using (SqlCommand cmd = new SqlCommand(usersSql, con))
          {
              cmd.Parameters.AddWithValue("@Search", searchText.Trim());

              using (SqlDataAdapter daUsers = new SqlDataAdapter(cmd))
              {
                  DataTable dtUsers = new DataTable();
                  daUsers.Fill(dtUsers);
                  dgvUsers.DataSource = dtUsers;
              }

          dgvUsers.SelectionMode = DataGridViewSelectionMode.FullRowSelect;

        private string GenerateBarcode13()
        {
            using (SqlConnection con = new SqlConnection(connStr))
            {
                con.Open();

                string sql = @"
SELECT ISNULL(MAX(CAST(Barcode AS BIGINT)), 2000000000000)
FROM dbo.tablename
WHERE Barcode IS NOT NULL
  AND Barcode NOT LIKE '%[^0-9]%';
";

                using (SqlCommand cmd = new SqlCommand(sql, con))
                {
                    long maxBarcode = Convert.ToInt64(cmd.ExecuteScalar());
                    long nextBarcode = maxBarcode + 1;

                    return nextBarcode.ToString();
                }
            }
        }

        private void btnAddProduct_Click(object sender, EventArgs e)
        {
            if (Session.CurrentRole != "Admin")
            {
                MessageBox.Show("Only Admin can add products.");
                return;
            }

            string name = txtProductName.Text.Trim();
            string desc = string.IsNullOrWhiteSpace(txtDescription.Text) ? null : txtDescription.Text.Trim();
            string barcode = txtBarcode.Text.Trim();

            if (string.IsNullOrWhiteSpace(barcode))
            {
                barcode = GenerateBarcode13();
            }
            else
            {
                if (barcode.Length != 13 || !long.TryParse(barcode, out _))
                {
                    MessageBox.Show("Barcode must be exactly 13 digits.");
                    return;
                }
            }

            if (string.IsNullOrWhiteSpace(name))
            {
                MessageBox.Show("Product Name is required.");
                return;
            }

            if (cmbType.SelectedValue == null)
            {
                MessageBox.Show("Please select a category/type.");
                return;
            }

            int typeId = Convert.ToInt32(cmbType.SelectedValue);
            string category = cmbType.Text.Trim();

            if (!decimal.TryParse(txtCostPrice.Text.Trim(), out decimal costPrice) ||
                !decimal.TryParse(txtPrice.Text.Trim(), out decimal price) ||
                !int.TryParse(txtQuantity.Text.Trim(), out int qty))
            {
                MessageBox.Show("Enter valid numbers for CostPrice, Price and Quantity.");
                return;
            }

            if (costPrice < 0 || price < 0 || qty < 0)
            {
                MessageBox.Show("CostPrice, Price and Quantity cannot be negative.");
                return;
            }

            using (SqlConnection con = new SqlConnection(connStr))
            {
                con.Open();

                string sql = @"
INSERT INTO dbo.tablename
(ProductName, TypeId, Category, CostPrice, Price, Quantity, Description, ImagePath, Barcode, IsActive, CreatedAt, UpdatedAt)
VALUES
(@Name, @TypeId, @Category, @CostPrice, @Price, @Qty, @Desc, @ImagePath, @Barcode, 1, GETDATE(), GETDATE());
";

                using (SqlCommand cmd = new SqlCommand(sql, con))
                {
                    cmd.Parameters.AddWithValue("@Name", name);
                    cmd.Parameters.AddWithValue("@TypeId", typeId);
                    cmd.Parameters.AddWithValue("@Category", category);
                    cmd.Parameters.AddWithValue("@CostPrice", costPrice);
                    cmd.Parameters.AddWithValue("@Price", price);
                    cmd.Parameters.AddWithValue("@Qty", qty);
                    cmd.Parameters.AddWithValue("@Desc", (object)desc ?? DBNull.Value);

                    cmd.Parameters.AddWithValue("@ImagePath",
                        string.IsNullOrWhiteSpace(selectedProductImageName)
                            ? (object)DBNull.Value : selectedProductImageName);

                    cmd.Parameters.AddWithValue("@Barcode", barcode);


                    cmd.ExecuteNonQuery();

                    LogProductActivity(con, null, "ADD_PRODUCT",
                        $"Added product '{name}' TypeId={typeId} Category='{category}' Price={price} Qty={qty}");
                }
            }

            MessageBox.Show("Product added successfully!");
            ClearInputs();
            LoadProducts();
            if (Session.CurrentRole == "Admin") LoadProductLogs();
        }

    private void ShowPetImageFromRuntime(string imageFileNameOrPath)
    {
        picPet.Image?.Dispose();
        picPet.Image = null;
        picPet.ImageLocation = null;

        if (string.IsNullOrWhiteSpace(imageFileNameOrPath))
            return;

        string fullPath = Path.IsPathRooted(imageFileNameOrPath)
            ? imageFileNameOrPath
            : Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "photos", "pets", imageFileNameOrPath);

        if (!File.Exists(fullPath))
            return;

        try
        {
            using (var fs = new FileStream(fullPath, FileMode.Open, FileAccess.Read, FileShare.ReadWrite))
            using (var img = Image.FromStream(fs))
            {
                picPet.Image = new Bitmap(img);
            }
        }
        catch
        {
            picPet.Image?.Dispose();
            picPet.Image = null;
        }
    }

        switch (range)
        {
            case "Today":
                from = today;
                to = today.AddDays(1);

            case "Yesterday":

                break;

            case "Last 30 Days":
                from = today.AddDays(-29);
                to = today.AddDays(1);
                break;

            case "This Month":
                from = new DateTime(today.Year, today.Month, 1);
                to = today.AddDays(1);
                break;

            case "Last Month":
                DateTime firstDayThisMonth = new DateTime(today.Year, today.Month, 1);
                DateTime firstDayLastMonth = firstDayThisMonth.AddMonths(-1);

                from = firstDayLastMonth;
                to = firstDayThisMonth;
                break;

            case "All":
            default:
                from = null;
                to = null;
                break;
        }

        LoadPetLogs(from, to);

    {
        if (!IsReception())
        {
            MessageBox.Show("Only Reception can check barcodes.");
            return;
        }

        if (_selectedOrderId <= 0)
        {
            MessageBox.Show("Please select an order first.");
            txtBarcode.Clear();
            return;
        }

        string barcode = txtBarcode.Text.Trim();

        if (string.IsNullOrWhiteSpace(barcode))
            return;

        if (dgvOrderItems.DataSource == null || dgvOrderItems.Rows.Count == 0)
        {
            MessageBox.Show("This order has no items.");
            txtBarcode.Clear();
            return;
        }

        bool found = false;
        string matchedProductName = "";

        foreach (DataGridViewRow row in dgvOrderItems.Rows)
        {
            if (row.IsNewRow) continue;

            string rowBarcode = row.Cells["Barcode"].Value?.ToString()?.Trim() ?? "";

            if (rowBarcode == barcode)
            {
                found = true;
                matchedProductName = row.Cells["ProductName"].Value?.ToString() ?? "Product";
                row.Selected = true;
                dgvOrderItems.CurrentCell = row.Cells[0];
                break;
            }
        }

        if (found)
        {
            MessageBox.Show(
                $"Matched: {matchedProductName}",
                "Barcode Check",
                MessageBoxButtons.OK,
                MessageBoxIcon.Information);
        }
        else
        {
            MessageBox.Show(
                "This barcode is not part of the selected order.",
                "Barcode Check",
                MessageBoxButtons.OK,
                MessageBoxIcon.Warning);
        }

        txtBarcode.Clear();
        txtBarcode.Focus();
    }

    private void LoadPets(string keyword = "", string ageFilter = "", string priceFilter = "")
    {
        flpCards.Controls.Clear();

        using (var con = new SqlConnection(_connStr))
        {
            string query = @"
SELECT PetId, PetName, Breed, Gender, AgeMonths, Price, Description, ImagePath
FROM table name
WHERE IsAvailable = 1 AND TypeId = @TypeId";

            if (!string.IsNullOrWhiteSpace(keyword))
                query += " AND PetName LIKE @Keyword";

            if (!string.IsNullOrWhiteSpace(ageFilter) && ageFilter != "All Ages")
            {
                if (ageFilter == "0-6 months")
                    query += " AND AgeMonths BETWEEN 0 AND 6";
                else if (ageFilter == "6-12 months")
                    query += " AND AgeMonths BETWEEN 6 AND 12";
                else if (ageFilter == "1+ year")
                    query += " AND AgeMonths >= 12";
                else if (ageFilter == "3+ years")
                    query += " AND AgeMonths >= 36";
            }

            if (!string.IsNullOrWhiteSpace(priceFilter) && priceFilter != "All Prices")
            {
                if (priceFilter == "Under $100")
                    query += " AND Price < 100";
                else if (priceFilter == "$100 - $300")
                    query += " AND Price BETWEEN 100 AND 300";
                else if (priceFilter == "$300 - $500")
                    query += " AND Price BETWEEN 300 AND 500";
                else if (priceFilter == "Above $500")
                    query += " AND Price > 500";
            }

            query += " ORDER BY PetId DESC";

            using (var cmd = new SqlCommand(query, con))
            {
                cmd.Parameters.AddWithValue("@TypeId", _typeId);
                if (!string.IsNullOrWhiteSpace(keyword))
                    cmd.Parameters.AddWithValue("@Keyword", "%" + keyword + "%");

                con.Open();
                using (var r = cmd.ExecuteReader())
                {
                    bool hasRows = false;

                    while (r.Read())
                    {
                        hasRows = true;

                        int id = Convert.ToInt32(r["PetId"]);
                        string name = Convert.ToString(r["PetName"]);
                        string breed = Convert.ToString(r["Breed"]);
                        string gender = Convert.ToString(r["Gender"]);
                        int ageMonths = r["AgeMonths"] == DBNull.Value ? 0 : Convert.ToInt32(r["AgeMonths"]);
                        decimal price = r["Price"] == DBNull.Value ? 0 : Convert.ToDecimal(r["Price"]);
                        string desc = Convert.ToString(r["Description"]);
                        string imagePath = Convert.ToString(r["ImagePath"]);

                        flpCards.Controls.Add(BuildPetCard(id, name, breed, gender, ageMonths, price, desc, imagePath));
                    }

                    if (!hasRows)
                    {
                        var lblEmpty = new Label
                        {
                            AutoSize = false,
                            Width = flpCards.Width - 30,
                            Height = 60,
                            TextAlign = ContentAlignment.MiddleCenter,
                            Font = new Font("Segoe UI", 12F, FontStyle.Bold),
                            ForeColor = Color.FromArgb(120, 80, 90),
                            Text = "No pets found for the selected filters."
                        };
                        flpCards.Controls.Add(lblEmpty);
                    }
                }
            }
        }
    }

    {

        cmbPriceFilter.Items.Add("$0 - $10");
        cmbPriceFilter.Items.Add("$10 - $25");
        cmbPriceFilter.Items.Add("$25 - $50");
        cmbPriceFilter.Items.Add("$50+");
        cmbPriceFilter.SelectedIndex = 0;

        cmbSortOrder.Items.Clear();
        cmbSortOrder.Items.Add("Latest → Oldest");
        cmbSortOrder.Items.Add("Oldest → Latest");
        cmbSortOrder.Items.Add("Price: Low → High");
        cmbSortOrder.Items.Add("Price: High → Low");
        cmbSortOrder.SelectedIndex = 0;

        cmbPriceFilter.SelectedIndexChanged += (s, e) => LoadProducts();
        cmbSortOrder.SelectedIndexChanged += (s, e) => LoadProducts();
    }

    private void LoadProducts()
    {
        flpCards.Controls.Clear();

        decimal? minPrice = null;
        decimal? maxPrice = null;

        string priceChoice = cmbPriceFilter.SelectedItem?.ToString() ?? "All Prices";
        if (priceChoice == "$0 - $10") { minPrice = 0; maxPrice = 10; }
        else if (priceChoice == "$10 - $25") { minPrice = 10; maxPrice = 25; }
        else if (priceChoice == "$25 - $50") { minPrice = 25; maxPrice = 50; }
        else if (priceChoice == "$50+") { minPrice = 50; maxPrice = null; }

        string sortChoice = cmbSortOrder.SelectedItem?.ToString() ?? "Latest → Oldest";
        string orderBy = "CreatedAt DESC";

        if (sortChoice == "Oldest → Latest") orderBy = "CreatedAt ASC";
        else if (sortChoice == "Price: Low → High") orderBy = "Price ASC";
        else if (sortChoice == "Price: High → Low") orderBy = "Price DESC";

        using (var con = new SqlConnection(_connStr))
        using (var cmd = new SqlCommand($@"
SELECT ProductId, ProductName, Category, Description, Price, Quantity, ImagePath
FROM Products
WHERE IsActive = 1 AND TypeId = @TypeId
AND (@MinPrice IS NULL OR Price >= @MinPrice)
AND (@MaxPrice IS NULL OR Price <= @MaxPrice)
ORDER BY {orderBy};", con))
        {
            cmd.Parameters.AddWithValue("@TypeId", _typeId);
            cmd.Parameters.AddWithValue("@MinPrice", (object)minPrice ?? DBNull.Value);
            cmd.Parameters.AddWithValue("@MaxPrice", (object)maxPrice ?? DBNull.Value);

            con.Open();
            using (var r = cmd.ExecuteReader())
            {
                while (r.Read())
                {
                    int id = Convert.ToInt32(r["ProductId"]);
                    string name = Convert.ToString(r["ProductName"]);
                    string category = Convert.ToString(r["Category"]);
                    string desc = Convert.ToString(r["Description"]);
                    decimal price = Convert.ToDecimal(r["Price"]);
                    int qty = Convert.ToInt32(r["Quantity"]);
                    string imagePath = Convert.ToString(r["ImagePath"]);

                    flpCards.Controls.Add(BuildProductCard(id, name, category, desc, price, qty, imagePath));
                }
            }
        }
    }

    private void btnConfirm_Click(object sender, EventArgs e)
    {
        string phone = txtPhone.Text.Trim();
        string address = txtAddress.Text.Trim();
        string notes = txtNotes.Text.Trim();

        if (!CartManager.Items.Any())
        {
            MessageBox.Show("Cart is empty.");
            return;
        }

        if (string.IsNullOrWhiteSpace(phone))
        {
            MessageBox.Show("Phone number is required.");
            return;
        }

        if (string.IsNullOrWhiteSpace(address))
        {
            MessageBox.Show("Address is required.");
            return;
        }

        int userId = Session.CurrentUserId;
        if (userId <= 0)
        {
            MessageBox.Show("Session expired. Please log in again.");
            return;
        }

        var productItems = CartManager.Items
            .Where(x => x.ItemType.Equals("Product", StringComparison.OrdinalIgnoreCase))
            .ToList();

        var petItems = CartManager.Items
            .Where(x => x.ItemType.Equals("Pet", StringComparison.OrdinalIgnoreCase))
            .ToList();

        decimal totalAmount = CartManager.Items.Sum(x => x.UnitPrice * x.Quantity);

        try
        {
            using (SqlConnection conn = new SqlConnection(DbHelper.ConnectionString))
            {
                conn.Open();
                using (SqlTransaction tx = conn.BeginTransaction())
                {
                    try
                    {
                        int orderId = 0;

                        if (productItems.Any())
                        {
                            orderId = OrderService.PlaceOrder(
                                userId, phone, address, notes, totalAmount, conn, tx);

                            foreach (var item in productItems)
                            {
                                OrderService.AddOrderItem(
                                    orderId,
                                    item.ItemId,
                                    item.Quantity,
                                    item.UnitPrice,
                                    conn,
                                    tx);

                                using (var cmd = new SqlCommand(
                                    @"UPDATE Products
                                      SET Quantity = Quantity - @Qty
                                      WHERE ProductId = @ProductId AND Quantity >= @Qty;", conn, tx))
                                {
                                    cmd.Parameters.AddWithValue("@Qty", item.Quantity);
                                    cmd.Parameters.AddWithValue("@ProductId", item.ItemId);

                                    int affected = cmd.ExecuteNonQuery();
                                    if (affected == 0)
                                        throw new Exception($"Product #{item.ItemId} is out of stock.");
                                }
                            }
                        }

                        foreach (var pet in petItems)
                        {
                            using (var cmd = new SqlCommand(
                                @"UPDATE Pets
                                  SET IsAvailable = 0
                                  WHERE PetId = @PetId AND IsAvailable = 1;", conn, tx))
                            {
                                cmd.Parameters.AddWithValue("@PetId", pet.ItemId);

                                int affected = cmd.ExecuteNonQuery();
                                if (affected == 0)
                                    throw new Exception($"Pet #{pet.ItemId} is no longer available.");
                            }
                        }

                        tx.Commit();

                        MessageBox.Show(
                            $"Checkout complete ✅\n\n" +
                            $"Products: {productItems.Count}\n" +
                            $"Pets: {petItems.Count}\n" +
                            $"Total: ${totalAmount:0.00}" +
                            (productItems.Any() ? $"\nOrder ID: #{orderId}" : ""),
                            "Success",
                            MessageBoxButtons.OK,
                            MessageBoxIcon.Information
                        );

                        foreach (var it in CartManager.Items.ToList())
                            CartManager.Remove(it.ItemType, it.ItemId);

                        this.Close();
                    }
                    catch
                    {
                        tx.Rollback();
                        throw;
                    }
                }
            }
        }
        catch (Exception ex)
        {
            MessageBox.Show("❌ Failed to checkout: " + ex.Message);
        }
    }

    {
        var row = _repo.GetUserById(userId);
        if (row == null)
        {
            MessageBox.Show("User not found.");
            return;
        }

        txtName.Text = row["Name"]?.ToString();
        txtEmail.Text = row["Email"]?.ToString();
        txtPhone.Text = row["PhoneNumber"]?.ToString();
        txtPassword.UseSystemPasswordChar = true;
        txtPassword.Text = row["Password"]?.ToString();
    }

    private void CheckReportReplies()
    {
        string email = txtEmail.Text.Trim();
        if (string.IsNullOrWhiteSpace(email)) return;

        bool hasReply = _reportRepo.HasRepliedReport(email);

        if (hasReply)
        {
            MessageBox.Show(
                "We saw your report. We will contact you as soon as possible.",
                "Report Update",
                MessageBoxButtons.OK,
                MessageBoxIcon.Information
            );
        }
    }

        LoadUserInfo(userId);
        CheckReportReplies();
    }

public partial class ReportForm : BaseForm

    private readonly ReportRepository _repo = new ReportRepository();

    {
        InitializeComponent();
        _email = email ?? "";
        _phone = phone ?? "";

        btnSend.Click -= btnSend_Click;
        btnSend.Click += btnSend_Click;

        this.Load += ReportForm_Load;
    }

    private void ReportForm_Load(object sender, EventArgs e)
    {
        cmbReason.Items.Clear();
        cmbReason.Items.AddRange(new object[]
        {
            "Bug / App not working",
            "Wrong info",
            "Payment / Checkout issue",
            "Suggestion",
            "Other"
        });
        cmbReason.SelectedIndex = 0;

        txtEmail.Text = _email;
        txtPhone.Text = _phone;
    }

    private void btnSend_Click(object sender, EventArgs e)
    {
        try
        {
            string email = txtEmail.Text.Trim();
            string phone = txtPhone.Text.Trim();
            string reason = cmbReason.SelectedItem?.ToString() ?? "Other";
            string message = txtMessage.Text.Trim();

            if (string.IsNullOrWhiteSpace(email))
            {
                MessageBox.Show("Email is required.");
                return;
            }

            if (string.IsNullOrWhiteSpace(message))
            {
                MessageBox.Show("Please write your report.");
                return;
            }

            int reportId = _repo.CreateReport(new Report
            {
                ReporterEmail = email,
                ReporterPhone = string.IsNullOrWhiteSpace(phone) ? null : phone,
                TargetType = "System",
                TargetId = 0,
                Reason = reason,
                Description = message
            });

            MessageBox.Show($"Report sent ✅ (ID #{reportId})");
            this.Close();
        }
        catch (Exception ex)
        {
            MessageBox.Show("Failed to send report:\n" + ex.Message);
        }
    }

    private void btnCancel_Click(object sender, EventArgs e)
    {
        this.Close();
    }

    private void btnBack_Click(object sender, EventArgs e)
    {
        ProfileForm frm = new ProfileForm();
        frm.Show();
        this.Close();
    }
}
